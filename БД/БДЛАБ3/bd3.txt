DROP TABLE IF EXISTS person CASCADE; 
DROP TABLE IF EXISTS emotion CASCADE;
DROP TABLE IF EXISTS environment CASCADE;
DROP TABLE IF EXISTS transformation CASCADE;
DROP TABLE IF EXISTS interaction CASCADE;
DROP TABLE IF EXISTS creation CASCADE;
DROP TABLE IF EXISTS petals CASCADE;
DROP TABLE IF EXISTS petalsType CASCADE;
DROP TABLE IF EXISTS activitydetails CASCADE;
DROP TABLE IF EXISTS reactionfromaction CASCADE;
DROP TABLE IF EXISTS conditionenvironment CASCADE;
DROP TABLE IF EXISTS creationtype CASCADE;


CREATE TABLE Person (
    person_id SERIAL PRIMARY KEY,
    name VARCHAR(20) NOT NULL,
    gender VARCHAR(1) NOT NULL
);

CREATE TABLE Emotion (
    emotion_id SERIAL PRIMARY KEY,
    type_emotion VARCHAR(20) NOT NULL,
    intensity INTEGER CHECK(intensity > 0 AND intensity < 11),
    person_id INTEGER NOT NULL REFERENCES person(person_id)
);

CREATE TABLE ConditionEnvironment (
    temperature FLOAT PRIMARY KEY,
    humidity FLOAT NOT NULL
);

CREATE TABLE Environment (
    environment_id SERIAL PRIMARY KEY,
    name_environment VARCHAR(20) NOT NULL,
    temperature FLOAT NOT NULL REFERENCES ConditionEnvironment(temperature)
);

CREATE TABLE CreationType (
    type_creation VARCHAR(30) PRIMARY KEY,
    size INT NOT NULL,
    color VARCHAR(30) NOT NULL
);

CREATE TABLE Creation (
    creation_id SERIAL PRIMARY KEY,
    type_creation VARCHAR(30) NOT NULL REFERENCES CreationType(type_creation) ,
    well_being VARCHAR(50) NOT NULL
);


CREATE TABLE transformation(
    transformation_id SERIAL PRIMARY KEY,
    state_changed BOOLEAN,
    reason VARCHAR(30) NOT NULL,
    environment_id INTEGER NOT NULL REFERENCES environment(environment_id),
    creation_id INTEGER NOT NULL REFERENCES creation(creation_id)
);

CREATE TABLE petalsType (
    type_petals VARCHAR(30) PRIMARY KEY,
    texture VARCHAR(20) NOT NULL
);

CREATE TABLE petals(
    petals_id SERIAL PRIMARY KEY,
    type_petals VARCHAR(30) NOT NULL REFERENCES petalsType(type_petals),
    modification VARCHAR(50) NOT NULL,
    creation_id INTEGER NOT NULL REFERENCES creation(creation_id)
);

CREATE TABLE reactionFromAction (
    reaction_from_action SERIAL PRIMARY KEY,
    reaction_creation VARCHAR(30) NOT NULL,
    reaction_person VARCHAR(30) NOT NULL
);

CREATE TABLE ActivityDetails (
    activity VARCHAR(50) PRIMARY KEY,
    duration INT CHECK(duration >= 0),
    distance INT CHECK(distance >= 0),
    reaction_from_action INT NOT NULL REFERENCES reactionFromAction(reaction_from_action)
);

CREATE TABLE interaction(
    interaction_id SERIAL PRIMARY KEY,
    activity VARCHAR(50) REFERENCES ActivityDetails(activity),
    person_id INTEGER NOT NULL REFERENCES person(person_id),
    creation_id INTEGER NOT NULL REFERENCES creation(creation_id)
);
