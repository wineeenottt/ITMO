ORG 0x4CA
ADR_RES: WORD $RES
READY: WORD 0x40
END_CHAR: WORD 0x0A

START: CLA

SYMBOL1: IN 5
		 AND READY
		 BEQ SYMBOL1
		 IN 4
		 CMP END_CHAR
		 BEQ EXIT
		 SWAB
		 ST (ADR_RES)
		 
SYMBOL2: IN 5
		 AND READY
		 BEQ SYMBOL2
		 LD (ADR_RES)
		 IN 4
		 CMP END_CHAR
		 BEQ EXIT
		 ST (ADR_RES)+
		 JUMP SYMBOL1
		 
EXIT:  ST (ADR_RES)
	   HLT
	  
ORG 0x56E
RES: WORD 0x00
